\begin{frame}
  \frametitle{rocFFT multi-GPU/multi-node API}

  \begin{center}
    Same API for single-proc/multi-proc
  \end{center}

  \bigskip
  
  \begin{itemize}
  \item Divide the input space into \lstinline{rocfft_brick}s using
    \lstinline{rocfft_brick_create}
    \begin{itemize}
    \item Bricks are explicitly associated to devices
    \item Bricks are implicitly associated to an MPI rank
    \end{itemize}
  \item Add the bricks to a \lstinline{rocfft_field} using
    \lstinline{rocfft_field_add_brick}
  \item Add the field to a \lstinline{rocfft_plan_description} using
    \lstinline{rocfft_plan_description_add_infield} or
    \lstinline{rocfft_plan_description_add_outfield}
  \item Add the description to the plan as usual and execute the plan.
  \end{itemize}
  
  \bigskip

  \begin{center}
    Single-GPU/multi-gpu/multi-process transforms use the same library.
  \end{center}
  
\end{frame}


\begin{frame}
  \frametitle{cuFFT API Comparison}


  \begin{itemize}
  \item Single-proc multi-gpu
    \begin{itemize}
    \item cuFFTXT interface.
    \item Only supports \lstinline{CUFFT_XT_FORMAT_INPLACE}
      \begin{itemize}
      \item 1D regular decomposition
      \end{itemize}
    \end{itemize}
  \item Multi-prod multi-gpu
    \begin{itemize}
    \item cuFFTMP interface.
    \item Built-in slabs: \lstinline{CUFFT_XT_FORMAT_INPLACE}
    \item Custom brick decomposition: \lstinline{CUFFT_XT_FORMAT_DISTRIBUTED_INPUT}
    \item real/complex transforms only support
      \lstinline{CUFFT_XT_FORMAT_DISTRIBUTED_INPUT}, which means that
      you get to set blocks via
      \lstinline{cufftMpMakePlanDecomposition} allows for a brick
      decomposition.
    \item It seems that only one buffer per rank is allowed.
    \item Makes use of nvshmem
    \end{itemize}
    \item NB: New NVIDIA ARM-based NVPL includes FFT in early access release.
  \end{itemize}

  
\end{frame}

\begin{frame}
  \frametitle{Communication Library Support}

  Compiling rocFFT should work with any GPU-aware MPI library.

  \medskip
  
  Tested with OpenMPI and Cray-MPI

  \bigskip
  
  Release plan:
  \begin{itemize}
  \item Source is always available:
    \begin{itemize}
    \item cmake option \lstinline{ROCFFT_MPI_ENABLE}
      \begin{itemize}
        \item \lstinline{ROCFFT_CRAY_MPI_ENABLE} for cray-mpi to link
          \lstinline{rtl}
      \end{itemize}
    \end{itemize}
  \item Planned OS packages: rocfft, rocfft-openmpi, rocfft-mpich
    \begin{itemize}
    \item rocFFT packages can only support MPI libraries libraries
      which also have OS packages
    \end{itemize}
  \end{itemize}
  \medskip

  
\end{frame}

\begin{frame}
  \frametitle{2D FFT regular decomposition}

  Input is decomposed in the $x$ direction, output in the $y$ direction.

  \medskip
  
  \begin{enumerate}
  \item Perform batched 1D FFTs in the $y$ direction
  \item Transpose the data (eg \texttt{MPI_Alltoall}) so that the $x$
    direction is contiguous
  \item Perform batched 1D FFTs in the $x$ direction
  \end{enumerate}

  \medskip

  \begin{itemize}
  \item Each device sends $P-1$ messages of size $N^2/P$.
  \end{itemize}
  
\end{frame}

\begin{frame}[fragile]
  \frametitle{3D FFT with 1D decomposition (``slab'')}
  
  Input is decomposed in the $x$ direction, output in the $z$ direction.

 
  \medskip
  
  \begin{enumerate}
  \item Perform batched 2D FFTs in the $z/y$ direction
  \item Transpose the data so that the $x$ direction is contiguous
  \item Perform batched 1D FFTs in the $x$ direction
  \end{enumerate}

  \medskip

  \begin{itemize}
  \item Each device sends $P-1$ messages of size $N^3/P$.
  \end{itemize}
    
\end{frame}

\begin{frame}[fragile]
  \frametitle{3D FFT with 1D decomposition (``slab'')}
  \begin{figure}[h]
    \begin{asy}
import three;

int N = 16;
int P = 4;
real w = 0.1;

size(200,0);

pen surfpen=lightgreen+opacity(0.5);
//draw(surface(g2), lightblue);

triple[] pblock = new triple[];
pblock.push((0, 0, 0));
pblock.push((0, N * w, 0));
pblock.push((0, N * w, N * w));
pblock.push((0, 0, N * w));
for(int idx = 0; idx < 4; ++ idx) {
    pblock.push(pblock[idx] + (w * (N # P), 0, 0));
}

for(int idx = 0; idx < pblock.length; ++idx) {
    //dot((string)idx, pblock[idx]);
}

path3[] sblock = new path3[];
sblock.append(pblock[0]--pblock[1]--pblock[2]--pblock[3]--cycle);
sblock.append(pblock[4]--pblock[5]--pblock[6]--pblock[7]--cycle);
sblock.append(pblock[0]--pblock[1]--pblock[5]--pblock[4]--cycle);
sblock.append(pblock[0]--pblock[1]--pblock[5]--pblock[4]--cycle);
sblock.append(pblock[0]--pblock[4]--pblock[7]--pblock[3]--cycle);
sblock.append(pblock[2]--pblock[3]--pblock[7]--pblock[6]--cycle);
sblock.append(pblock[1]--pblock[2]--pblock[6]--pblock[5]--cycle);

for(int pidx = 0; pidx < P; ++pidx) {
    for(int idx = 0; idx < sblock.length; ++idx) {
        triple offset  = (pidx * (1 + N # P) * w, 0 , 0);
            
        draw(surface(shift(offset) * sblock[idx]), Pen(pidx));
    }
}

      
    \end{asy}
  \end{figure}
\end{frame}

\begin{frame}[fragile]
  \frametitle{3D FFT with 2D decomposition (``pencil'')}
  
  Input is decomposed in the $x/y$ direction, output in the $y/z$ direction.

 
  \medskip
  
  \begin{enumerate}
  \item Perform batched 1D FFTs in the $z$ direction
  \item Transpose the data so that the $y$ direction is contiguous
  \item Perform batched 1D FFTs in the $y$ direction
  \item Transpose the data so that the $x$ direction is contiguous
  \item Perform batched 1D FFTs in the $x$ direction
  \end{enumerate}

  \medskip

  \begin{itemize}
  \item Two communication stages
  \item In each stage, each device sends $\sqrt{P}-1$ messages of size $N^3/P$.
  \end{itemize}
      
\end{frame}

\begin{frame}[fragile]
  \frametitle{3D FFT with 2D decomposition (``pencil'')}
  \begin{figure}[h]
    \begin{asy}
      import three;

int N = 16;
int P = 2;
real w = 0.1;

size(200,0);

pen surfpen=lightgreen+opacity(0.5);
//draw(surface(g2), lightblue);

triple[] pblock = new triple[];
pblock.push((0, 0, 0));
pblock.push((0, (N # P) * w, 0));
pblock.push((0, (N # P) * w, N * w));
pblock.push((0, 0, N * w));
for(int idx = 0; idx < 4; ++ idx) {
    pblock.push(pblock[idx] + (w * (N # P), 0, 0));
}

for(int idx = 0; idx < pblock.length; ++idx) {
    //dot((string)idx, pblock[idx]);
}

path3[] sblock = new path3[];
sblock.append(pblock[0]--pblock[1]--pblock[2]--pblock[3]--cycle);
sblock.append(pblock[4]--pblock[5]--pblock[6]--pblock[7]--cycle);
sblock.append(pblock[0]--pblock[1]--pblock[5]--pblock[4]--cycle);
sblock.append(pblock[0]--pblock[1]--pblock[5]--pblock[4]--cycle);
sblock.append(pblock[0]--pblock[4]--pblock[7]--pblock[3]--cycle);
sblock.append(pblock[2]--pblock[3]--pblock[7]--pblock[6]--cycle);
sblock.append(pblock[1]--pblock[2]--pblock[6]--pblock[5]--cycle);

for(int pxidx = 0; pxidx < P; ++pxidx) {
    for(int pyidx = 0; pyidx < P; ++pyidx) {
        int pidx = pyidx + P * pxidx;
        for(int idx = 0; idx < sblock.length; ++idx) {
            triple offset  = (pxidx * (1 + N # P) * w,
                              pyidx * (1 + N # P) * w ,
                              0);


            draw(surface(shift(offset) * sblock[idx]), Pen(pidx));
        }
    }
}
    \end{asy}
  \end{figure}
\end{frame}

\begin{frame}
  \frametitle{Prioritization}

  \begin{center}
    Q: Lots of work to be done: what should be done first?
  \end{center}

  \pause
  
  \begin{center}
    A: Testing infrastructure.  \pause But then?
  \end{center}

  \pause
  
  \begin{enumerate}
  \item Irregular data decompositions
    \begin{itemize}
    \item First pass: move data to a regular decomposition
    \item Second pass: heuristic to reduce communication costs
    \item Third pass: heuristic to reduce transform time
    \end{itemize}
  \item Sparse data
    \begin{itemize}
    \item Some bricks could have all-zero data.
    \end{itemize}
  \item Hybrid parallelism
    \begin{itemize}
    \item Multiple GPUs per rank is an optimisation target
    \end{itemize}
  \item Communication libraries
    \begin{itemize}
    \item rocSHMEM (large effort for rocFFT)
    \item RCCL
    \item MPI extensions (eg: streams)
    \end{itemize}
  \end{enumerate}
  
\end{frame}
