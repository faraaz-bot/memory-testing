\section{Introduction}

This document contains notes for computing real-to-real FFTs (like
sine transforms) for \texttt{rocFFT}.


As way of reference, the FFT of complex inputs $\{x_n\}_{n=0}^{N-1}$
is
\begin{dmath}
  X_k = \sum_{k=0}^{N-1} x_n \omega_N^{nk}
\end{dmath}
with $\omega_N=e^{\frac{-2 \pi i}{N}}$.


\section{DST}


\subsection{DST-I}


Given $\{x_n\}_{n=0}^{N-1}$,
\begin{dmath}
  X_k = \sum_{n=0}^{N-1}x_n \sin\left[\frac{\pi}{N+1}(n+1)(k+1)\right]
\end{dmath}
It would appear that this can be computed from a FFT of length $N$
with pre/post-processing.

This is equivalent to the negative imaginary part of the DFT of the
length-$2N+2$ sequence
\begin{dmath}
  \tilde{x}
  = \{0, x_0, x_1, \dots, x_{N-1}, 0, -x_{N-1}, \dots, -x_1, -x_0\}
  = \left\{
  \begin{array}{ll}
    0, & n = 0\\
    x_{n-1}, & n = 1, \dots, N\\
    0, & n = N + 1\\
    -x_{2N-n+1}, & n = N+2, \dots, 2N +1
  \end{array}
  \right.
\end{dmath}
and taking output indices $1,\dots,N$.

Consider
\begin{dmath}
  \sum_{n=0}^{2N+1} \tilde{x}_n \omega_{2N+2}^{nk}
  =
  \cancelto{0}{\tilde{x}_0 \omega_{2N+2}^{nk}}\quad
  + \sum_{n=1}^{N} \tilde{x}_n \omega_{2N+2}^{nk}
  + \cancelto{0}{\tilde{x}_{N+1} \omega_{2N+2}^{(N+1)k}}\quad
  + \sum_{n=N+2}^{2N+1} \tilde{x}_n \omega_{2N+2}^{nk}
  =
  \sum_{n=1}^{N} x_{n-1} \omega_{2N+2}^{nk}
  + \sum_{n=N+2}^{2N+1} x_{2N-n+1} \omega_{2N+2}^{nk}
  =
  \sum_{n=0}^{N-1} x_{n} \omega_{2N+2}^{(n+1)k}
  + \sum_{n=0}^{N-1} -x_{N -n -1} \omega_{2N+2}^{(n + N + 2)k}
  =
  \sum_{n=0}^{N-1} \(\omega_{2N+2}^{k}x_n - \omega_{2N+2}^{N +2k} x_{N-n-1}  \) \omega_{2N+2}^{nk}
\end{dmath}


\subsection{DST-II}

\begin{dmath}
  X_k = \sum_{n=0}^{N-1}x_n \sin\left[\frac{\pi}{N+1}\(n+\frac{1}{2}\)(k+1)\right]
\end{dmath}

This transform can be compuated via a length-$4N$ transform with input
\begin{dmath}
  \tilde{x} = \{0, x_0, 0, x_1, 0, \dots, x_{N-1} ,0,  -x_{N-1}, 0, -x_{N-2}\dots , 0,  -x_2, 0, -x_1 \}
\end{dmath}
That is, for $0\leq n < N$,
\begin{eqnarray}
  \tilde{x}_n &= \tilde{x}_{4N - (2n +1)} = 0 \\
  \tilde{x}_{2n + 1} &= x_n \\
  \tilde{x}_{4N - (2n + 1)} &= -x_n.
\end{eqnarray}

Consider
\begin{dmath}
  \sum_{n=0}^{4N-1} \tilde{x}_n \omega_{4N}^{nk}
  =
  \cancelto{0}{\sum_{n=0}^{2N-1} \tilde{x}_{2n} \omega_{4N}^{2nk}}\quad
  +
  \sum_{n=0}^{2N-1} \tilde{x}_{2n+1} \omega_{4N}^{(2n+1)k}
  =
  \sum_{n=0}^{N-1} \tilde{x}_{2n+1} \omega_{4N}^{(2n+1)k}
  +
  \sum_{n=0}^{N-1} \tilde{x}_{2n+2N+1} \omega_{4N}^{(2n+2N+1)k}
  =
  \sum_{n=0}^{N-1} x_n \omega_{4N}^{(2n+1)k}
  +
  \sum_{n=0}^{N-1} -x_{N -n -1} \omega_{4N}^{(2n+2N+1)k}
  =
  \omega_{4N}^{k} \sum_{n=0}^{N-1} x_n \omega_{2N}^{nk} 
  +
  \cancelto{\(-1\)^k}{\omega_{2}^{k} }\qquad  \omega_{4N}^{k}
  \sum_{n=0}^{N-1} -x_{N -n -1} \omega_{2N}^{nk}
  =
  \omega_{4N}^{k}
  \(
  \sum_{n=0}^{N-1} \(x_n -\(-1\)^k x_{N-n-1} \) \omega_{2N}^{nk}
  \).
\end{dmath}
To transform $\omega_{2N}$ to $\omega_{N}$, we divide the sum into
even and odd parts.  Q: what to do about the $(-1)^k$?
