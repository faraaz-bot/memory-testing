\begin{frame}
  \frametitle{Spectral Ops}

  \begin{center}
    Spectral ops are about \emph{data flow}.
  \end{center}
    
  \medskip
  
  Two categories:
  \begin{itemize}
  \item One-way
    \begin{itemize}
      \item $
    \text{input}
    \rightarrow \text{read op}
    \rightarrow \mathcal{F}
    \rightarrow \text{write op}
    \rightarrow \text{output}
    $
  \item examples: shift, window, filter, scale
    \end{itemize}

  \item round-trip\\
    \begin{itemize}
    \item$
      \text{input}
      \rightarrow \text{read op}
      \rightarrow \mathcal{F}
      \rightarrow \text{mid op}
      \rightarrow \mathcal{F}^{-1}
      \rightarrow \text{write op}
      \rightarrow \text{output}
      $
      \item examples: Poisson solve, convolution, correlation
    \end{itemize}
  \end{itemize}

\end{frame}


\begin{frame}
  \frametitle{Spectral Ops: One-way Examples}

  \begin{itemize}
  \item Window filter
    \begin{itemize}
    \item Input: real 1D array.
    \item Read-op: $f_{i_x} \rightarrow H(i_x) f_{i_x}$
    \item $H$ is a Hamming window, Hann windows, Blackman windows, \dots
    \item Use case: DSP
    \end{itemize}

  \item Phase-shift-and-scale
    \begin{itemize}
    \item Input: multiple 1D complex arrays of length $N$
    \item Each 1D array is shifted by phase $\theta(b)$ and scaled by $\lambda(b)$
    \item Read op: $X_{i_x,b} \rightarrow \omega_N^{i_x \theta(b)} \lambda(b)X_{i_x,b}$
    \item Use case: radio astronomy.
    \end{itemize}
  %% \item $y$ derivative of a 2D Hermitian array
  %%   \begin{itemize}
  %%   \item Read op: $X_{i_x, i_Y,b} \rightarrow \frac{2 \pi}{L_y} i_y  X_{i_x,b}$
  %%   \end{itemize}
  \item $x$ derivative of a 2D Hermitian array
    \begin{itemize}
    \item Define $$k_x= \frac{2 \pi}{L_x}\begin{cases}i_x &\text{ if } i_x < N_x / 2\\ N_x -i_x &\text{otherwise}\end{cases} $$
    \item Read op: $X_{i_x, i_Y,b} \rightarrow i k_x  X_{i_x,b}$
    \end{itemize}
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Spectral Ops: Round-Trip Examples}

  \begin{itemize}

   \item Poisson solve
    \begin{itemize}
    \item Round-trip, input: Real array (2D, $N_x\times{}N_y$)
    \item Output: find $f$ such that $\nabla^2{f} = \phi$
    \item Mid op: $x_{i_x, i_y} \rightarrow \frac{-x_{i_x, i_y} }{ k_x^2 + k_y^2}$
    \item Maybe a constant scale as well
    \item Use case: molecular dynamics
    \end{itemize}
  \item Phase-shift dealiased convolution for Navier--Stokes
    \begin{itemize}
    \item Input: Three Hermitian 3D arrays, cubic
    \item Read op:
      \begin{itemize}
      \item Read-op:
        $u
          \rightarrow \(u, \omega = \nabla \times u\)
          \rightarrow \(u, \omega, \tilde{u}, \tilde{\omega}\)
          $
        \item $\tilde{u}$ is $u$ with a $\(1/2, 1/2, 1/2\)$ phase shift.
      \end{itemize}
    \item Mid-op: two copies of 
      \begin{dmath}
        \(f_0, f_1, f_2, f_3, f_4, f_5\)
        \rightarrow
        \(
        f_4 f_2 - f_5 f_1,
        f_5 f_1 - f_3 f_2,
        f_3 f_1 - f_4 f_0
        \)
      \end{dmath}
    \item write-op: combine-and-zero
      \begin{itemize}
      \item $\(f_0, f_1, f_2, f_3, f_4, f_5\)
        \rightarrow
        \(f_0 + f_1, f_2 + f_3, f_4 + f_5 \)$
      \item All values with $\left|k\right| > N$ set to zero.
      \end{itemize}
    \item Use-case: GESTS (they also add a passive scalar field).
    \end{itemize}
  \end{itemize}
  
\end{frame}

\begin{frame}
  \frametitle{Spectral Ops: convolution examples}

  These examples need zero-padding to dealias convolutions.
  
  \begin{itemize}

  \item Source term for vorticity Navier--Stokes 2D
    \begin{itemize}
      \item Input: 2D Hermitian array
      \item Read-op: $\omega \rightarrow
        \(
        ik_x\omega,
        \frac{ik_x}{k_x^2 + k_y^2}\omega,
        ik_y\omega,
        \frac{ik_y}{k_x^2 + k_y^2}\omega
        \)$
      \item mid-op: $\(f_0, f_1, f_2, f_3\) \rightarrow \( f_3 f_0 - f_2 f_0 \)$
    \end{itemize}
  \item Convolution for ML
    \begin{itemize}
    \item Mid-op: $\(f,g\)\rightarrow fg$
    \item $f$ is a constant filter: only transform once, keep data for later mid-ops.
    \end{itemize}
  \item Dealiased autocorrelation
    \begin{itemize}
    \item mid-op: input multiplied by its complex conjugate: $f \rightarrow \bar{f}f$
    \end{itemize}
  \end{itemize}
  
\end{frame}


\begin{frame}
  \frametitle{Spectral Ops: ops}

  \medskip
  Operations are:
  \begin{itemize}
  \item local (single index)
  \item can have many inputs
  \item can have many outputs
  \item depend on index and batch number
  \item have user-supplied constants
  \item quadratic mid-ops may require dealiasing (ie conv)
  \end{itemize}

  \medskip

  There are lots of possible operations, and combinations thereof.

  \medskip

  Not all spectral ops will apply to all fields; we may have some
  depth-1 DAG associated with input/output fields for each spectral
  op.

\end{frame}
  
\begin{frame}
  \frametitle{Spectral Ops: inputs and outputs}
  
  \medskip
  Input and outputs
  \begin{itemize} 
  \item may have different types
    \begin{itemize}
      \item float, int4, double, \dots
    \end{itemize}
  \item may have different lengths
    \begin{itemize}
    \item seemingly only for round-trip?
    \end{itemize}
  \item some data may be constant across executions
    \begin{itemize}
    \item eg: filters, ML weights, Bluestein method
    \end{itemize}
  \end{itemize}
  
  
\end{frame}


\begin{frame}
  \frametitle{Multi-GPU}

  \begin{center}
    Multi-GPU is about \emph{data distribution}.
  \end{center}

  \medskip

  \begin{itemize}
  \item Single-node and multi-node.
    \begin{itemize}
    \item single-node doesn't need a communicator
    \item multi-node needs a communicator
    \end{itemize}
  \item brick decomposition
    \begin{itemize}
    \item pencil and slab are just degenerate cases
    \end{itemize}
  \item MPI and rocSHMEM back-ends
  \item Decomposition should work also work with the normal interface
    \begin{itemize}
    \item This allows for sparsity in the single-GPU case.
    \end{itemize}
  \end{itemize}
  
\end{frame}


\begin{frame}
  \frametitle{Sparsity}

  \begin{center}
    Sparsity is about \emph{data properties}.
  \end{center}

  
  Given the brick decomposition in the multi-GPU section, we can mark
  bricks as containing only zero data.

  \medskip

  \begin{itemize}
  \item bricks can be marked as sparse, ie all zero.
  \item Or we can add compression!  Perhaps lossy!
  \item Separate sparsity for inputs and output.
  \item Brick-based sparsity design
    \begin{itemize}
    \item spectral data seem to have coarser sparsity patterns
    \item csr or coordinate list doesn't seem relevant
    \end{itemize}
  \end{itemize}
      
\end{frame}

\begin{frame}
  \frametitle{Spectral ops API proposals}

  Callbacks are absurdly slow in ROCm.
  \begin{itemize}
  \item NVIDIA has moved to JIT to improve callback performance (\href{https://www.nvidia.com/en-us/on-demand/session/gtcfall21-a31155/?playlistId=playList-ead11304-9931-4e91-9d5a-fb0e1ef27014}{link})
  \item Can we make use of RTC to do this in rocFFT?
  \end{itemize}
  
  Spectral ops API proposals:
  \begin{itemize}
    \item Separate APIs per op:
    \begin{itemize}
    \item \href{https://github.com/ROCmSoftwarePlatform/rocFFT-internal/pull/1122}{Pr 1122}
    \item Easier to use.
    \item Restricts user interaction with internals.
    \item Ops are bespoke: if we don't provide it, the user can't use it.
    \end{itemize}
  \item RTC-based general API:
    \begin{itemize}
    \item \href{https://github.com/ROCmSoftwarePlatform/rocFFT-internal/pull/1119}{PR 1119}
    \item Similar complexity to callbacks.
    \item Minor interaction with internals
    \item Users can add their own functionality.
    \item (Can we just extend callbacks?)
    \end{itemize}
  \end{itemize}


  
\end{frame}

\begin{frame}
  \frametitle{Multi-GPU, sparsity API proposals}

  \href{https://github.com/ROCmSoftwarePlatform/rocFFT-internal/pull/1119}{PR 1119} also provides multi-GPU and sparse functionality.

  \begin{itemize}
  \item A new \texttt{rocfft_field} struct is added.
  \item Fields contain:
    \begin{itemize}
    \item Precision/type information
    \item Brick decomposition information
    \end{itemize}
  \item Multiple fields can be added to a \texttt{rocfft_plan_description}
  \item Descriptors are (optionally) passed to a \texttt{rocfft_plan}.
  \item Plans are executed, performing the FFT.
  \end{itemize}

     
\end{frame}

\begin{frame}
  \frametitle{Multi-GPU, sparsity API proposals}

  For spectral ops, fields are used to
  \begin{itemize}
  \item Set the number of inputs and outputs for the op.
  \item Set which particular fields are touched by the op.
  \item Set the per-field precision and data type.
  \end{itemize}
  
  If no fields are present, execution defaults to a single input and a
  single output.

  \medskip

  For multi-GPU and sparsity, fields are used to
  \begin{itemize}
  \item to coordinate the data decomposition
    \begin{itemize}
    \item Bricks are added to a field
    \item Each brick has a device ID, a rank, and a sparsity/compression value.
    \end{itemize}
  \end{itemize}
  
  
\end{frame}



\begin{frame}
  \frametitle{Multi-GPU, sparsity API proposals}

  \medskip
  If fields are specified, then \texttt{rocfft_execute} takes fields
  passed as \texttt{void **}.
  \begin{itemize}
  \item Fields also store (user-specified) GPU buffer pointers.
  \end{itemize}
  Alternatively, we just let the user specify the array of
  (rank-local) pointers in the execution function.
  

  \bigskip

  Internally, all data is a \texttt{rocfft_precision}; our kernels
  handle only specific data types.

  \medskip
  For internal computation, we can
  \begin{itemize}
  \item specify the per-transform precision (for multi-field spectral ops)
  \item allow for lossy data compression / type change for multi-GPU
    communication.
  \end{itemize}

  \end{frame}

\begin{frame}
  \frametitle{Other Notable APIs}

  Spectral ops:
  \begin{itemize}
  \item \texttt{cuFFTDx}: allows users to call \texttt{cuFFT} from
    their kernels.
  \item \texttt{vkFFT}: Convolution matrix, symmetry, fused kernels.
  \item \texttt{OneAPI}: Some c2c convolutions for ML.
  \item \texttt{FFTW++}: pointer-based spectral ops for convolutions.
  \item \texttt{safari}: binary convolution with normalized spec op for ML.
  \end{itemize}

  \medskip

  Multi-GPU:
  \begin{itemize}
  \item \texttt{cuFFTXt}: single-node, only slab decomposition.
  \item \texttt{cuFFTMp}: brick decomposition.
  \item \texttt{heFFTe}: brick decomposition.
  \item \texttt{FFTW++}: pencil decomposition.
  \item \texttt{SpFFT}: also sparsity.
  \end{itemize}
    
\end{frame}

\begin{frame}
  \frametitle{Optional Additions}

  Besides the basic API design shown above, we have more optional
  functionality that we can add:

  \medskip
  
  \begin{itemize}
  \item Multi-transform specops can have different precisions per transform
  \item Multi-value spectral ops can have a field-dependancy list.
  \item Multi-GPU transforms can have lossy compression for communication.
  \item Different fields can have different lengths
  \item Convolution outputs can have different lengths than inputs.
  \end{itemize}
  
  
\end{frame}
