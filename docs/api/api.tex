\subsubsection{Sample application}
\begin{lstlisting}
// Transform parameters
std::vector<size_t> length = {8, 8};
std::vector<size_t> devices = {0, 1};
std::vector<std::vector<size_t>> ilower = {{0,0},{0,4}};
std::vector<std::vector<size_t>> iupper = {{4,8},{8,8}};
std::vector<size_t> istride = {1,8};
std::vector<std::vector<size_t>> olower = {{0,0},{4,0}};
std::vector<std::vector<size_t>> oupper = {{8,4},{8,8}};
std::vector<size_t> ostride = {8,1};

// Set up field and for input:
rocfft_field ifield = nullptr;
rocfft_field_create(&ifield);
std::vector<rocfft_brick> iwall(devices.size());
for(int idx = 0; idx < devices.size(); ++idx) {
  rocfft_brick_create(&iwall[idx],
                      ilower[idx].data(), iupper[idx].data(),
                      istride.data(),   istride.size();
                      devices[idx]);
  rocfft_field_add_brick(infield, iwall[idx]);
  rocfft_brick_destroy(iwall[idx]);
}

// Set up field and bricks for output:
rocfft_field ofield = nullptr;
rocfft_field_create(&ofield);
std::vector<rocfft_brick> owall(devices.size());
for(int idx = 0; idx < devices.size(); ++idx) {
  rocfft_brick_create(&owall[idx],
                      olower[idx].data(), oupper[idx].data(),
                      istride.data(),   ostride.size();
                      devices[idx]);
                      rocfft_field_add_brick(outfield, owall[idx]);
  rocfft_brick_destroy(owall[idx]);
}

// Set up description and plan:
rocfft_plan_description description = nullptr;
rocfft_plan_description_create(&description);
// TODO: set data layout needed?
rocfft_plan_description_add_infield(description, ifield);
rocfft_plan_description_add_outfield(description, ofield);
rocfft_plan gpu_plan = nullptr;
rocfft_plan_create(&gpu_plan,
                  rocfft_placement_notinplace,
                  rocfft_transform_type_complex_forward,
                  rocfft_precision_double,
                  length.size(),
                  length.data(),
                  1,
                  description);

//Execution stage
std::vector<void*> ibuf(devices.size());
std::vector<void*> obuf(devices.size());
const size_t bufsize = sizeof(std::complex<double> * 4 * 8);
for(int idx = 0; idx < devices.size(); ++idx) {
  hipmalloc(&ibuf[idx], bufsize);
  hipmalloc(&obuf[idx], bufsize);
}
rocfft_execute(gpu_plan, (void**)ibuf.data(), (void**)obuf.data(), null);
\end{lstlisting}

// Transform parameters:  
std::vector<size_t> length = {8, 8};

const int N0 = 8;
const double Lx = 2.0, Ly = 2.0; // Physical constants
const double kx = 2 * pi * (idx0 < N0/2 ? idx0 : N0 - idx0) / Lx;
const double ky = 2 * pi * idx1 / L1;

out0 = in0 * I * kx * in0;
out1 = in0 * I * ky * in0;
";

rocfft_plan_description_create(&description);
rocfft_plan_description_set_writep(description, writeop);
